// ================= MAIN DO POST HANDLER =================
function doPost(e) {
    try {
        let params = {};
        if (e && e.postData && e.postData.contents) {
            try {
                params = JSON.parse(e.postData.contents);
            } catch (parseError) {
                const content = e.postData.contents;
                const urlParams = new URLSearchParams(content);
                for (const [key, value] of urlParams) {
                    params[key] = value;
                }
            }
        }
        
        const action = params.action || 'ADD_RECEIPT';
        
        if (action === 'ADD_RECEIPT') {
            return handleAddReceipt(params);
        } else if (action === 'GET_CATEGORIES') {
            return handleGetCategories();
        } else if (action === 'GET_YEARS') {
            return handleGetYears();
        } else if (action === 'GET_UPI_ID') {
            return handleGetUpiId();
        } else {
            return createResponse({
                success: false,
                error: 'Unknown action: ' + action
            });
        }
        
    } catch (error) {
        console.error('Error in doPost:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE ADD RECEIPT =================
function handleAddReceipt(params) {
    try {
        const requiredFields = [
            'branchId', 'homeId', 'receiptNo',
            'receiptCategory', 'financialYear', 'amount', 'receiptAmount'
        ];

        for (const field of requiredFields) {
            if (!params[field] && params[field] !== 0) {
                return createResponse({
                    success: false,
                    error: 'Missing required field: ' + field
                });
            }
        }

        const ss = SpreadsheetApp.getActiveSpreadsheet();
        let sheet = ss.getSheetByName('Data');

        if (!sheet) {
            sheet = ss.insertSheet('Data');
        }

        ensureDataHeaders(sheet);

        const now = new Date();
        const timestamp = Utilities.formatDate(
            now,
            Session.getScriptTimeZone(),
            'yyyy-MM-dd HH:mm:ss'
        );
        const dateStr = Utilities.formatDate(
            now,
            Session.getScriptTimeZone(),
            'yyyy-MM-dd'
        );

        let screenshotUrl = '';
        let ocrText = '';
        let ocrTransactionId = '';
        let ocrAmount = '';
        let ocrDate = '';
        let ocrUpiId = '';
        let ocrStatus = '';

        // Process payment screenshot when supplied.
        if (params.screenshotBase64) {
            try {
                const proof = saveAndReadPaymentScreenshot(
                    params.screenshotBase64,
                    params.screenshotName || ('payment-' + params.receiptNo + '.jpg'),
                    params.screenshotMimeType || 'image/jpeg'
                );

                screenshotUrl = proof.screenshotUrl || '';
                ocrText = proof.ocrText || '';
                ocrTransactionId = proof.transactionId || '';
                ocrAmount = proof.amount || '';
                ocrDate = proof.date || '';
                ocrUpiId = proof.upiId || '';
                ocrStatus = proof.ocrStatus || '';
            } catch (proofError) {
                console.error('Screenshot processing error:', proofError);
                ocrStatus = 'Screenshot saved, OCR failed: ' + proofError.toString();
            }
        }

        // Prefer the transaction ID manually entered by the user.
        const finalTransactionId =
            String(params.transactionId || '').trim() || ocrTransactionId;

        const rowData = [
            params.branchId || '',
            params.homeId || '',
            params.receiptNo || '',
            finalTransactionId,
            params.receiptCategory || '',
            params.financialYear || '',
            params.senderUpiId || ocrUpiId || '',
            parseFloat(params.amount) || 0,
            parseFloat(params.receiptAmount) || 0,
            parseFloat(params.balance) || 0,
            'Verification Pending',
            params.remarks || '',
            timestamp,
            dateStr
        ];

        // Keep the original 14 columns unchanged.
        sheet.appendRow(rowData);

        // Add proof/OCR information to columns after the original columns.
        ensureProofColumns(sheet);

        const newRow = sheet.getLastRow();
        const proofValues = [
            params.paymentMethod || 'UPI',
            screenshotUrl,
            ocrTransactionId,
            ocrAmount,
            ocrDate,
            ocrUpiId,
            ocrStatus,
            ocrText
        ];

        sheet.getRange(newRow, 15, 1, proofValues.length).setValues([proofValues]);

        return createResponse({
            success: true,
            message: 'Payment receipt submitted successfully',
            receiptNo: params.receiptNo,
            status: 'Verification Pending',
            transactionId: finalTransactionId,
            screenshotUrl: screenshotUrl,
            ocr: {
                transactionId: ocrTransactionId,
                amount: ocrAmount,
                date: ocrDate,
                upiId: ocrUpiId,
                status: ocrStatus
            }
        });

    } catch (error) {
        console.error('Error in handleAddReceipt:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

function ensureDataHeaders(sheet) {
    const headers = [
        'Branch ID', 'Home ID', 'Receipt No', 'Transaction ID',
        'Receipt Category', 'Financial Year', 'Sender UPI ID',
        'Amount', 'Receipt Amount', 'Balance', 'Payment Confirm', 'Remarks',
        'Timestamp', 'Date'
    ];

    if (sheet.getLastRow() === 0) {
        sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
        return;
    }

    const current = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), headers.length))
        .getValues()[0];

    let changed = false;
    headers.forEach((header, index) => {
        if (!current[index]) {
            sheet.getRange(1, index + 1).setValue(header);
            changed = true;
        }
    });

    return changed;
}

function ensureProofColumns(sheet) {
    const proofHeaders = [
        'Payment Method',
        'Screenshot URL',
        'OCR Transaction ID',
        'OCR Amount',
        'OCR Date',
        'OCR UPI ID',
        'OCR Status',
        'OCR Text'
    ];

    proofHeaders.forEach((header, i) => {
        const col = 15 + i;
        const existing = sheet.getRange(1, col).getValue();
        if (!existing) {
            sheet.getRange(1, col).setValue(header);
        }
    });
}

/**
 * Saves the original screenshot to Drive and attempts OCR.
 *
 * IMPORTANT:
 * Enable the Advanced Google service "Drive API" in Apps Script:
 * Services (+) -> Drive API -> Add.
 * Also ensure the Drive API is enabled in the linked Google Cloud project.
 */
function saveAndReadPaymentScreenshot(base64, fileName, mimeType) {
    if (!base64) {
        throw new Error('Empty screenshot.');
    }

    // Safety limit: approximately 4 MB decoded image.
    const bytes = Utilities.base64Decode(base64);
    if (bytes.length > 4 * 1024 * 1024) {
        throw new Error('Screenshot is larger than 4 MB.');
    }

    const safeName = sanitizeFileName(fileName);
    const blob = Utilities.newBlob(bytes, mimeType, safeName);

    const folder = getOrCreatePaymentProofFolder();
    const originalFile = folder.createFile(blob);
    originalFile.setDescription('UPI payment screenshot submitted from member e-pay page.');

    let ocrText = '';
    let ocrStatus = 'OCR unavailable';

    try {
        // Advanced Drive API required.
        const ocrResource = {
            title: 'OCR-' + safeName,
            mimeType: MimeType.GOOGLE_DOCS
        };

        const ocrFile = Drive.Files.insert(ocrResource, blob, {
            ocr: true,
            convert: true,
            ocrLanguage: 'en'
        });

        if (ocrFile && ocrFile.id) {
            Utilities.sleep(1200);
            ocrText = DocumentApp.openById(ocrFile.id).getBody().getText() || '';
            ocrStatus = ocrText ? 'OCR successful' : 'OCR returned no text';

            // Delete temporary OCR document after reading it.
            try {
                Drive.Files.remove(ocrFile.id);
            } catch (deleteError) {
                console.warn('Could not remove OCR temp file:', deleteError);
            }
        }
    } catch (ocrError) {
        console.error('OCR error:', ocrError);
        ocrStatus = 'OCR failed: ' + ocrError.toString();
    }

    const extracted = extractPaymentData(ocrText);

    return {
        screenshotUrl: originalFile.getUrl(),
        ocrText: ocrText,
        transactionId: extracted.transactionId,
        amount: extracted.amount,
        date: extracted.date,
        upiId: extracted.upiId,
        ocrStatus: ocrStatus
    };
}

function getOrCreatePaymentProofFolder() {
    const folderName = 'UPI Payment Screenshots';

    const folders = DriveApp.getFoldersByName(folderName);
    if (folders.hasNext()) {
        return folders.next();
    }

    return DriveApp.createFolder(folderName);
}

function sanitizeFileName(name) {
    return String(name || 'payment-screenshot.jpg')
        .replace(/[\\\/:*?"<>|#%{}[\]]/g, '_')
        .substring(0, 150);
}

function extractPaymentData(text) {
    const clean = String(text || '')
        .replace(/\r/g, '\n')
        .replace(/[ \t]+/g, ' ')
        .trim();

    if (!clean) {
        return {
            transactionId: '',
            amount: '',
            date: '',
            upiId: ''
        };
    }

    let transactionId = '';

    // Look specifically near common UPI transaction/reference labels first.
    const transactionPatterns = [
        /(?:UTR|UPI\s*REF(?:ERENCE)?|TRANSACTION\s*ID|TRANSACTION\s*NO|REF(?:ERENCE)?\s*NO)\s*[:#-]?\s*([A-Z0-9]{6,30})/i,
        /(?:RRN)\s*[:#-]?\s*([A-Z0-9]{6,30})/i
    ];

    for (const pattern of transactionPatterns) {
        const match = clean.match(pattern);
        if (match && match[1]) {
            transactionId = match[1].trim();
            break;
        }
    }

    let amount = '';
    const amountMatch = clean.match(
        /(?:AMOUNT|PAID|TOTAL|DEBITED|SENT)\s*[:\-]?\s*(?:₹|INR|RS\.?)?\s*([\d,]+(?:\.\d{1,2})?)/i
    );

    if (amountMatch) {
        amount = amountMatch[1].replace(/,/g, '');
    } else {
        const rupeeMatch = clean.match(/(?:₹|INR|RS\.?)\s*([\d,]+(?:\.\d{1,2})?)/i);
        if (rupeeMatch) {
            amount = rupeeMatch[1].replace(/,/g, '');
        }
    }

    // Extract a VPA-like value. This can be the payer or payee depending on the screenshot.
    const upiMatch = clean.match(/\b[A-Z0-9._-]{2,}@[A-Z0-9._-]{2,}\b/i);
    const upiId = upiMatch ? upiMatch[0] : '';

    let date = '';
    const dateMatch = clean.match(
        /\b(\d{1,2}[\/.-]\d{1,2}[\/.-]\d{2,4}(?:\s+\d{1,2}:\d{2}(?::\d{2})?\s*(?:AM|PM)?)?)\b/i
    );

    if (dateMatch) {
        date = dateMatch[1];
    }

    return {
        transactionId: transactionId,
        amount: amount,
        date: date,
        upiId: upiId
    };
}

// ================= HANDLE GET CATEGORIES =================
function handleGetCategories() {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Receipt Category and Amount');
        
        if (!sheet) {
            const newSheet = ss.insertSheet('Receipt Category and Amount');
            newSheet.getRange(1, 1, 1, 3).setValues([['Receipt ID', 'Receipt Category', 'Receipt Amount']]);
            newSheet.getRange(2, 1, 2, 3).setValues([
                ['AP', 'Annual Pay', 500],
                ['YN', 'Yoganadham', 500]
            ]);
            
            return createResponse({
                success: true,
                categories: [
                    { id: 'AP', category: 'Annual Pay', amount: 500 },
                    { id: 'YN', category: 'Yoganadham', amount: 500 }
                ]
            });
        }
        
        const data = sheet.getDataRange().getValues();
        const categories = [];
        
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][0] && data[i][1]) {
                categories.push({
                    id: data[i][0].toString().trim(),
                    category: data[i][1].toString().trim(),
                    amount: parseFloat(data[i][2]) || 0
                });
            }
        }
        
        if (categories.length === 0) {
            return createResponse({
                success: true,
                categories: [
                    { id: 'AP', category: 'Annual Pay', amount: 500 },
                    { id: 'YN', category: 'Yoganadham', amount: 500 }
                ]
            });
        }
        
        return createResponse({
            success: true,
            categories: categories
        });
        
    } catch (error) {
        console.error('Error in handleGetCategories:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE GET FINANCIAL YEARS =================
function handleGetYears() {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Financial Years');
        
        if (!sheet) {
            const newSheet = ss.insertSheet('Financial Years');
            newSheet.getRange(1, 1, 1, 1).setValues([['Financial Year']]);
            newSheet.getRange(2, 1, 2, 1).setValues([
                ['2024-25'],
                ['2025-26']
            ]);
            
            return createResponse({
                success: true,
                years: ['2024-25', '2025-26']
            });
        }
        
        const data = sheet.getDataRange().getValues();
        const years = [];
        
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][0]) {
                const year = data[i][0].toString().trim();
                if (year) {
                    years.push(year);
                }
            }
        }
        
        if (years.length === 0) {
            return createResponse({
                success: true,
                years: ['2024-25', '2025-26']
            });
        }
        
        return createResponse({
            success: true,
            years: years
        });
        
    } catch (error) {
        console.error('Error in handleGetYears:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE GET UPI ID =================
function handleGetUpiId() {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Receipt UPI ID');
        
        if (!sheet) {
            const newSheet = ss.insertSheet('Receipt UPI ID');
            newSheet.getRange(1, 1, 1, 1).setValues([['Receipt UPI ID']]);
            newSheet.getRange(2, 1, 1, 1).setValues([['gokulg.ambadi@okicici']]);
            
            return createResponse({
                success: true,
                upiId: 'gokulg.ambadi@okicici'
            });
        }
        
        const data = sheet.getDataRange().getValues();
        let upiId = '';
        
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][0]) {
                const value = data[i][0].toString().trim();
                if (value) {
                    upiId = value;
                    break;
                }
            }
        }
        
        if (!upiId) {
            return createResponse({
                success: true,
                upiId: 'gokulg.ambadi@okicici'
            });
        }
        
        return createResponse({
            success: true,
            upiId: upiId
        });
        
    } catch (error) {
        console.error('Error in handleGetUpiId:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= CREATE RESPONSE =================
function createResponse(data) {
    const response = ContentService.createTextOutput();
    response.setMimeType(ContentService.MimeType.JSON);
    response.setContent(JSON.stringify(data));
    return response;
}

// ================= DO GET HANDLER =================
function doGet(e) {
    const action = e.parameter.action || 'info';
    
    if (action === 'info') {
        return createResponse({
            success: true,
            message: 'Receipt API is running',
            version: '1.0',
            endpoints: [
                'POST - ADD_RECEIPT',
                'GET - GET_CATEGORIES',
                'GET - GET_YEARS',
                'GET - GET_UPI_ID'
            ]
        });
    } else if (action === 'GET_CATEGORIES') {
        return handleGetCategories();
    } else if (action === 'GET_YEARS') {
        return handleGetYears();
    } else if (action === 'GET_UPI_ID') {
        return handleGetUpiId();
    } else {
        return createResponse({
            success: false,
            error: 'Unknown action: ' + action
        });
    }
}

// ================= SHEET CREATION HELPER =================
function createRequiredSheets() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetNames = ['Data', 'Receipt Category and Amount', 'Financial Years', 'Receipt UPI ID'];
    
    sheetNames.forEach(name => {
        if (!ss.getSheetByName(name)) {
            ss.insertSheet(name);
        }
    });
    
    const dataSheet = ss.getSheetByName('Data');
    const dataHeaders = [
        'Branch ID', 'Home ID', 'Receipt No', 'Transaction ID', 
        'Receipt Category', 'Financial Year', 'Sender UPI ID', 
        'Amount', 'Receipt Amount', 'Balance', 'Payment Confirm', 'Remarks',
        'Timestamp', 'Date'
    ];
    dataSheet.getRange(1, 1, 1, dataHeaders.length).setValues([dataHeaders]);
    
    const categorySheet = ss.getSheetByName('Receipt Category and Amount');
    const categoryData = categorySheet.getDataRange().getValues();
    if (categoryData.length === 0 || categoryData[0][0] !== 'Receipt ID') {
        categorySheet.getRange(1, 1, 1, 3).setValues([['Receipt ID', 'Receipt Category', 'Receipt Amount']]);
        if (categoryData.length <= 1) {
            categorySheet.getRange(2, 1, 2, 3).setValues([
                ['AP', 'Annual Pay', 500],
                ['YN', 'Yoganadham', 500]
            ]);
        }
    }
    
    const yearSheet = ss.getSheetByName('Financial Years');
    const yearData = yearSheet.getDataRange().getValues();
    if (yearData.length === 0 || yearData[0][0] !== 'Financial Year') {
        yearSheet.getRange(1, 1, 1, 1).setValues([['Financial Year']]);
        if (yearData.length <= 1) {
            yearSheet.getRange(2, 1, 2, 1).setValues([
                ['2024-25'],
                ['2025-26']
            ]);
        }
    }
    
    const upiSheet = ss.getSheetByName('Receipt UPI ID');
    const upiData = upiSheet.getDataRange().getValues();
    if (upiData.length === 0 || upiData[0][0] !== 'Receipt UPI ID') {
        upiSheet.getRange(1, 1, 1, 1).setValues([['Receipt UPI ID']]);
        if (upiData.length <= 1) {
            upiSheet.getRange(2, 1, 1, 1).setValues([['']]);
        }
    }
    
    return {
        success: true,
        message: 'All sheets created successfully',
        sheets: sheetNames
    };
}
