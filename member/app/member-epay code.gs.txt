function doPost(e) {
  const corsHeaders = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Access-Control-Max-Age": "86400"
  };
  
  try {
    let requestData;
    if (e.postData.type === "application/json") {
      requestData = JSON.parse(e.postData.contents);
    } else {
      requestData = JSON.parse(e.postData.getDataAsString());
    }
    
    console.log("Request received:", requestData);
    
    const action = requestData.action;
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    let result;
    switch(action) {
      case "ADD_RECEIPT":
        result = handleAddReceipt(ss, requestData);
        break;
      case "UPDATE_PAYMENT_STATUS":
        result = handleUpdatePaymentStatus(ss, requestData);
        break;
      default:
        result = { status: "error", error: "Unknown action: " + action };
    }
    
    return createResponse(result, corsHeaders);
    
  } catch(error) {
    console.error("Error in doPost:", error);
    return createResponse({ 
      status: "error", 
      error: error.toString(),
      stack: error.stack 
    }, corsHeaders);
  }
}

function doGet(e) {

// ==================== HANDLE ADD RECEIPT ====================
function handleAddReceipt(ss, data) {
  try {
    const sheet = ss.getSheetByName("Data");
    
    if (!sheet) {
      throw new Error("Data sheet not found");
    }
    
    // Ensure all columns exist (A to O)
    const headers = sheet.getRange(1, 1, 1, 15).getValues()[0];
    if (headers.length < 15) {
      const newHeaders = [
        "Branch ID", "Home ID", "Receipt No", "Transaction ID", 
        "Receipt Category", "Financial Year", "Sender UPI ID",
        "Amount", "Receipt Amount", "Balance", "Payment Confirm",
        "Remarks", "Timestamp", "Date", "Screenshot Data URL"
      ];
      sheet.getRange(1, 1, 1, newHeaders.length).setValues([newHeaders]);
    }
    
    const timestamp = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      'yyyy-MM-dd HH:mm:ss'
    );
    const dateStr = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      'yyyy-MM-dd'
    );
    
    // Handle screenshot data - store if present
    let screenshotData = '';
    if (data.screenshotDataUrl && data.screenshotDataUrl.startsWith('data:image/jpeg;base64,')) {
      screenshotData = data.screenshotDataUrl;
    } else if (data.screenshotDataUrl) {
      screenshotData = data.screenshotDataUrl;
    }
    
    // Prepare row data (15 columns)
    const rowData = [
      data.branchId || '',
      data.homeId || '',
      data.receiptNo || '',
      data.transactionId || '',
      data.receiptCategory || '',
      data.financialYear || '',
      data.senderUpiId || '',
      data.amount || 0,
      data.receiptAmount || 0,
      data.balance || 0,
      'Pending Verification',  // Column K: Payment Confirm
      data.remarks || '',
      timestamp,
      dateStr,
      screenshotData  // Column O: Screenshot Data URL
    ];
    
    // Append row to sheet
    const lastRow = sheet.getLastRow();
    sheet.getRange(lastRow + 1, 1, 1, rowData.length).setValues([rowData]);
    
    return { 
      status: "success", 
      message: "Receipt saved successfully",
      receiptNo: data.receiptNo,
      rowIndex: lastRow + 1,
      timestamp: timestamp,
      screenshotStored: !!screenshotData
    };
    
  } catch(error) {
    console.error("Error in handleAddReceipt:", error);
    throw error;
  }
}

// ==================== HANDLE UPDATE PAYMENT STATUS ====================
function handleUpdatePaymentStatus(ss, data) {
  try {
    const sheet = ss.getSheetByName("Data");
    
    if (!sheet) {
      throw new Error("Data sheet not found");
    }
    
    const rowIndex = data.rowIndex;
    if (!rowIndex && rowIndex !== 0) {
      throw new Error("Row index missing for update");
    }
    
    const rowNumber = parseInt(rowIndex);
    const lastRow = sheet.getLastRow();
    if (rowNumber > lastRow) {
      throw new Error("Row does not exist. Row number: " + rowNumber + ", Last row: " + lastRow);
    }
    
    const status = data.status;
    if (!status) {
      throw new Error("Status is required");
    }
    
    const notes = data.notes || '';
    const verifiedBy = data.verifiedBy || Session.getActiveUser().getEmail() || 'Admin';
    
    // Update Payment Confirm - Column K (index 11)
    sheet.getRange(rowNumber, 11).setValue(status);
    
    // Update Remarks with verification notes - Column L (index 12)
    if (notes) {
      const currentRemarks = sheet.getRange(rowNumber, 12).getValue() || '';
      const timestamp = Utilities.formatDate(
        new Date(),
        Session.getScriptTimeZone(),
        'dd/MM/yyyy HH:mm'
      );
      const newEntry = `[${timestamp}] ${status}: ${notes} (by: ${verifiedBy})`;
      const newRemarks = currentRemarks 
        ? currentRemarks + ' | ' + newEntry 
        : newEntry;
      sheet.getRange(rowNumber, 12).setValue(newRemarks);
    }
    
    // Update verification log
    try {
      updateVerificationLog(ss, data.receiptNo, status, notes, verifiedBy);
    } catch(logError) {
      console.error("Error updating verification log:", logError);
    }
    
    const verificationTime = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      'yyyy-MM-dd HH:mm:ss'
    );
    
    return { 
      status: "success", 
      message: "Payment status updated successfully",
      receiptNo: data.receiptNo,
      status: status,
      rowIndex: rowNumber,
      updatedAt: verificationTime
    };
    
  } catch(error) {
    console.error("Error in handleUpdatePaymentStatus:", error);
    throw error;
  }
}

// ==================== UPDATE VERIFICATION LOG ====================
function updateVerificationLog(ss, receiptNo, status, notes, verifiedBy) {
  try {
    let logSheet = ss.getSheetByName("Verification_Log");
    
    if (!logSheet) {
      logSheet = ss.insertSheet("Verification_Log");
      logSheet.getRange(1, 1, 1, 6).setValues([
        ['Receipt No', 'Status', 'Notes', 'Verified By', 'Timestamp', 'Date']
      ]);
    }
    
    const timestamp = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      'yyyy-MM-dd HH:mm:ss'
    );
    const dateStr = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      'yyyy-MM-dd'
    );
    
    logSheet.appendRow([
      receiptNo || '',
      status || '',
      notes || '',
      verifiedBy || '',
      timestamp,
      dateStr
    ]);
    
  } catch(error) {
    console.error("Error updating verification log:", error);
    throw error;
  }
}

// ==================== CREATE RESPONSE ====================
function createResponse(data, corsHeaders) {
  const output = ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
  
  if (corsHeaders) {
    for (const key in corsHeaders) {
      output.setHeader(key, corsHeaders[key]);
    }
  }
  
  return output;
}

function cleanupOldScreenshots() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Data");
  
  if (!sheet) return "Data sheet not found";
  
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return "No data to clean up";
  
  const keepRows = 100;
  const startRow = Math.max(2, lastRow - keepRows + 1);
  
  for (let i = 2; i < startRow; i++) {
    sheet.getRange(i, 15).setValue('');
  }
  
  return `Cleaned up screenshot data. Kept rows ${startRow} to ${lastRow}`;
}
