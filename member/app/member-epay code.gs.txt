/**
 * Google Apps Script - Receipt Submission Handler
 * Deploy as web app with "Execute as: Me" and "Who has access: Anyone"
 * 
 * All configuration is read from the spreadsheet dynamically
 */

// ================= MAIN DO POST HANDLER =================
function doPost(e) {
    try {
        // Parse the incoming request
        let params = {};
        if (e && e.postData && e.postData.contents) {
            try {
                params = JSON.parse(e.postData.contents);
            } catch (parseError) {
                // If JSON parsing fails, try URL encoded
                const content = e.postData.contents;
                const urlParams = new URLSearchParams(content);
                for (const [key, value] of urlParams) {
                    params[key] = value;
                }
            }
        }
        
        // Check for action
        const action = params.action || 'ADD_RECEIPT';
        
        if (action === 'ADD_RECEIPT') {
            return handleAddReceipt(params);
        } else if (action === 'GET_CATEGORIES') {
            return handleGetCategories();
        } else if (action === 'GET_YEARS') {
            return handleGetYears();
        } else if (action === 'GET_UPI_ID') {
            return handleGetUpiId();
        } else {
            return createResponse({
                success: false,
                error: 'Unknown action: ' + action
            });
        }
        
    } catch (error) {
        console.error('Error in doPost:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE ADD RECEIPT =================
function handleAddReceipt(params) {
    try {
        // Validate required fields
        const requiredFields = ['branchId', 'homeId', 'receiptNo', 'receiptCategory', 'financialYear', 'amount', 'receiptAmount'];
        for (const field of requiredFields) {
            if (!params[field] && params[field] !== 0) {
                return createResponse({
                    success: false,
                    error: 'Missing required field: ' + field
                });
            }
        }
        
        // Open the spreadsheet
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        let sheet = ss.getSheetByName('Data');
        
        if (!sheet) {
            // Create the sheet if it doesn't exist
            sheet = ss.insertSheet('Data');
            const headers = [
                'Branch ID', 'Home ID', 'Receipt No', 'Transaction ID', 
                'Receipt Category', 'Financial Year', 'Sender UPI ID', 
                'Amount', 'Receipt Amount', 'Balance', 'Payment Confirm', 'Remarks',
                'Timestamp', 'Date'
            ];
            sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
        }
        
        // Get current timestamp
        const now = new Date();
        const timestamp = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
        const dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd');
        
        // Prepare row data - Payment Confirm is "Verification Pending" by default
        const rowData = [
            params.branchId || '',
            params.homeId || '',
            params.receiptNo || '',
            params.transactionId || '',
            params.receiptCategory || '',
            params.financialYear || '',
            params.senderUpiId || '',
            parseFloat(params.amount) || 0,
            parseFloat(params.receiptAmount) || 0,
            parseFloat(params.balance) || 0,
            'Verification Pending', // Default confirmation status
            params.remarks || '',
            timestamp,
            dateStr
        ];
        
        // Append the row
        sheet.appendRow(rowData);
        
        // ========== CHECK AND UPDATE RECEIPT CATEGORY SHEET ==========
        // This ensures new categories are added dynamically
        try {
            const categorySheet = ss.getSheetByName('Receipt Category and Amount');
            if (categorySheet) {
                const categoryData = categorySheet.getDataRange().getValues();
                let categoryExists = false;
                
                // Check if category already exists (by name)
                for (let i = 1; i < categoryData.length; i++) {
                    if (categoryData[i] && categoryData[i][1] === params.receiptCategory) {
                        categoryExists = true;
                        break;
                    }
                }
                
                // If category doesn't exist, add it
                if (!categoryExists && params.receiptCategory) {
                    // Generate a simple ID from the category name
                    const newId = params.receiptCategory.substring(0, 3).toUpperCase() || 'NEW';
                    categorySheet.appendRow([newId, params.receiptCategory, parseFloat(params.amount) || 0]);
                }
            }
        } catch (e) {
            console.warn('Could not update category sheet:', e);
        }
        
        // ========== CHECK AND UPDATE FINANCIAL YEAR SHEET ==========
        try {
            const yearSheet = ss.getSheetByName('Financial Years');
            if (yearSheet) {
                const yearData = yearSheet.getDataRange().getValues();
                let yearExists = false;
                
                // Check if year already exists
                for (let i = 1; i < yearData.length; i++) {
                    if (yearData[i] && yearData[i][0] === params.financialYear) {
                        yearExists = true;
                        break;
                    }
                }
                
                // If year doesn't exist, add it
                if (!yearExists && params.financialYear) {
                    yearSheet.appendRow([params.financialYear]);
                }
            }
        } catch (e) {
            console.warn('Could not update financial year sheet:', e);
        }
        
        return createResponse({
            success: true,
            message: 'Receipt added successfully',
            receiptNo: params.receiptNo,
            status: 'Verification Pending'
        });
        
    } catch (error) {
        console.error('Error in handleAddReceipt:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE GET CATEGORIES =================
function handleGetCategories() {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Receipt Category and Amount');
        
        if (!sheet) {
            // Create default sheet if it doesn't exist
            const newSheet = ss.insertSheet('Receipt Category and Amount');
            newSheet.getRange(1, 1, 1, 3).setValues([['Receipt ID', 'Receipt Category', 'Receipt Amount']]);
            // Add default categories
            newSheet.getRange(2, 1, 2, 3).setValues([
                ['AP', 'Annual Pay', 500],
                ['YN', 'Yoganadham', 500]
            ]);
            
            return createResponse({
                success: true,
                categories: [
                    { id: 'AP', category: 'Annual Pay', amount: 500 },
                    { id: 'YN', category: 'Yoganadham', amount: 500 }
                ]
            });
        }
        
        const data = sheet.getDataRange().getValues();
        const categories = [];
        
        // Start from row 2 (skip header)
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][0] && data[i][1]) {
                categories.push({
                    id: data[i][0].toString().trim(),
                    category: data[i][1].toString().trim(),
                    amount: parseFloat(data[i][2]) || 0
                });
            }
        }
        
        // If no categories found, return defaults
        if (categories.length === 0) {
            return createResponse({
                success: true,
                categories: [
                    { id: 'AP', category: 'Annual Pay', amount: 500 },
                    { id: 'YN', category: 'Yoganadham', amount: 500 }
                ]
            });
        }
        
        return createResponse({
            success: true,
            categories: categories
        });
        
    } catch (error) {
        console.error('Error in handleGetCategories:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= HANDLE GET FINANCIAL YEARS =================
function handleGetYears() {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Financial Years');
        
        if (!sheet) {
            // Create default sheet if it doesn't exist
            const newSheet = ss.insertSheet('Financial Years');
            newSheet.getRange(1, 1, 1, 1).setValues([['Financial Year']]);
            newSheet.getRange(2, 1, 2, 1).setValues([
                ['2024-25'],
                ['2025-26']
            ]);
            
            return createResponse({
                success: true,
                years: ['2024-25', '2025-26']
            });
        }
        
        const data = sheet.getDataRange().getValues();
        const years = [];
        
        // Start from row 2 (skip header)
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][0]) {
                const year = data[i][0].toString().trim();
                if (year) {
                    years.push(year);
                }
            }
        }
        
        // If no years found, return defaults
        if (years.length === 0) {
            return createResponse({
                success: true,
                years: ['2024-25', '2025-26']
            });
        }
        
        return createResponse({
            success: true,
            years: years
        });
        
    } catch (error) {
        console.error('Error in handleGetYears:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= CREATE RESPONSE =================
function createResponse(data) {
    const response = ContentService.createTextOutput();
    response.setMimeType(ContentService.MimeType.JSON);
    response.setContent(JSON.stringify(data));
    return response;
}

// ================= DO GET HANDLER (for testing) =================
function doGet(e) {
    const action = e.parameter.action || 'info';
    
    if (action === 'info') {
        return createResponse({
            success: true,
            message: 'Receipt API is running',
            version: '1.0',
            endpoints: [
                'POST - ADD_RECEIPT (adds receipt with "Verification Pending" status)',
                'GET - GET_CATEGORIES (fetches from "Receipt Category and Amount" sheet)',
                'GET - GET_YEARS (fetches from "Financial Years" sheet)',
                'GET - GET_UPI_ID (fetches from "Receipt UPI ID" sheet)'

        return handleGetCategories();
    } else if (action === 'GET_YEARS') {
        return handleGetYears();
    } else if (action === 'GET_UPI_ID') {
        return handleGetUpiId();
    } else {
        return createResponse({
            success: false,
            error: 'Unknown action: ' + action
        });
    }
}

// ================= SHEET CREATION HELPERS =================
function createRequiredSheets() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetNames = ['Data', 'Receipt Category and Amount', 'Financial Years', 'Receipt UPI ID'];
    
    sheetNames.forEach(name => {
        if (!ss.getSheetByName(name)) {
            ss.insertSheet(name);
        }
    });
    
    // Setup Data sheet headers
    const dataSheet = ss.getSheetByName('Data');
    const dataHeaders = [
        'Branch ID', 'Home ID', 'Receipt No', 'Transaction ID', 
        'Receipt Category', 'Financial Year', 'Sender UPI ID', 
        'Amount', 'Receipt Amount', 'Balance', 'Payment Confirm', 'Remarks',
        'Timestamp', 'Date'
    ];
    dataSheet.getRange(1, 1, 1, dataHeaders.length).setValues([dataHeaders]);
    
    // Setup Receipt Category sheet headers
    const categorySheet = ss.getSheetByName('Receipt Category and Amount');
    const categoryData = categorySheet.getDataRange().getValues();
    if (categoryData.length === 0 || categoryData[0][0] !== 'Receipt ID') {
        categorySheet.getRange(1, 1, 1, 3).setValues([['Receipt ID', 'Receipt Category', 'Receipt Amount']]);
        // Add default categories only if no data exists
        if (categoryData.length <= 1) {
            categorySheet.getRange(2, 1, 2, 3).setValues([
                ['AP', 'Annual Pay', 500],
                ['YN', 'Yoganadham', 500]
            ]);
        }
    }
    
    // Setup Financial Years sheet headers
    const yearSheet = ss.getSheetByName('Financial Years');
    const yearData = yearSheet.getDataRange().getValues();
    if (yearData.length === 0 || yearData[0][0] !== 'Financial Year') {
        yearSheet.getRange(1, 1, 1, 1).setValues([['Financial Year']]);
        if (yearData.length <= 1) {
            yearSheet.getRange(2, 1, 2, 1).setValues([
                ['2024-25'],
                ['2025-26']
            ]);
        }
    }
    
    // Setup Receipt UPI ID sheet headers
    const upiSheet = ss.getSheetByName('Receipt UPI ID');
    const upiData = upiSheet.getDataRange().getValues();
    if (upiData.length === 0 || upiData[0][0] !== 'Receipt UPI ID') {
        upiSheet.getRange(1, 1, 1, 1).setValues([['Receipt UPI ID']]);
        if (upiData.length <= 1) {
            upiSheet.getRange(2, 1, 1, 1).setValues([['place upi id']]);
        }
    }
    
    return {
        success: true,
        message: 'All sheets created successfully',
        sheets: sheetNames
    };
}

// ================= UTILITY: GET SHEET DATA AS JSON =================
function getSheetData(sheetName, includeHeaders = true) {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName(sheetName);
        
        if (!sheet) {
            return { success: false, error: 'Sheet not found: ' + sheetName };
        }
        
        const data = sheet.getDataRange().getValues();
        
        if (data.length === 0) {
            return { success: true, data: [], headers: [] };
        }
        
        const headers = data[0];
        const rows = [];
        
        for (let i = 1; i < data.length; i++) {
            const row = {};
            for (let j = 0; j < headers.length; j++) {
                row[headers[j]] = data[i][j] || '';
            }
            rows.push(row);
        }
        
        return {
            success: true,
            headers: headers,
            data: rows,
            count: rows.length
        };
        
    } catch (error) {
        return {
            success: false,
            error: error.toString()
        };
    }
}

// ================= UTILITY: UPDATE CONFIRMATION STATUS =================
function updateConfirmationStatus(receiptNo, status) {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName('Data');
        
        if (!sheet) {
            return createResponse({
                success: false,
                error: 'Data sheet not found'
            });
        }
        
        const data = sheet.getDataRange().getValues();
        const confirmColumn = 11; // Column K (Payment Confirm)
        
        // Search for the receipt (Column C is Receipt No)
        for (let i = 1; i < data.length; i++) {
            if (data[i] && data[i][2] === receiptNo) {
                // Update the confirmation status
                sheet.getRange(i + 1, confirmColumn + 1).setValue(status);
                return createResponse({
                    success: true,
                    message: 'Confirmation status updated',
                    receiptNo: receiptNo,
                    status: status
                });
            }
        }
        
        return createResponse({
            success: false,
            error: 'Receipt not found: ' + receiptNo
        });
        
    } catch (error) {
        console.error('Error in updateConfirmationStatus:', error);
        return createResponse({
            success: false,
            error: error.toString()
        });
    }
}

// ================= CORS HEADERS (for additional support) =================
function setCorsHeaders(response) {
    response.setHeader('Access-Control-Allow-Origin', '*');
    response.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS');
    response.setHeader('Access-Control-Allow-Headers', 'Content-Type');
    return response;
}

// ================= OPTIONS HANDLER (for preflight requests) =================
function doOptions(e) {
    const response = ContentService.createTextOutput();
    response.setMimeType(ContentService.MimeType.TEXT);
    response.setContent('OK');
    return setCorsHeaders(response);
}
